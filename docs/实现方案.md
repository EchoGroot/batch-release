// 批次滚动更新

    // 1. 检查是否是最后一批

    // 1. 检查是否还有下一批

}

### 3.5 阶段五：ReplicaSet 管理核心（第 7 周）

#### 3.5.1 核心算法

```go
// 扩缩容协调
func (r *BatchReleaseReconciler) reconcileReplicaSets(
    ctx context.Context,
    br *v1alpha1.BatchRelease,
    deploy *appsv1.Deployment,
    newRS *appsv1.ReplicaSet,
    oldRSs []*appsv1.ReplicaSet,
) error {
    // 1. 计算当前批次目标
    batchGoal := calculateBatchGoal(br, deploy)
    
    // 2. 计算 MaxSurge 和 MaxUnavailable
    maxSurge := calculateMaxSurge(deploy)
    maxUnavailable := calculateMaxUnavailable(deploy)
    
    // 3. 计算新 RS 目标副本数
    newRSTarget := min(batchGoal, deploy.Spec.Replicas)
    
    // 4. 扩容新 RS
    if newRS.Spec.Replicas < newRSTarget {
        // 考虑 MaxSurge 限制
        scaleTo := min(newRSTarget, currentTotal - deploy.Spec.Replicas + maxSurge)
        scaleReplicaSet(newRS, scaleTo)
    }
    
    // 5. 缩容旧 RS
    if newRS.Status.AvailableReplicas >= newRSTarget {
        totalOldReplicas := getTotalReplicas(oldRSs)
        if totalOldReplicas > 0 {
            // 考虑 MaxUnavailable 限制
            scaleDownOldReplicaSets(oldRSs, maxUnavailable)
        }
    }
}
```

#### 3.5.2 优先级策略

旧 ReplicaSet 缩容优先级：
1. Replicas = 0 的 RS（已经缩容的）
2. Available Replicas = 0 的 RS（没有可用 Pod）
3. 创建时间最早的 RS

### 3.6 阶段六：Webhook 验证（第 8 周）

#### 3.6.1 ValidatingWebhook

**文件**: `pkg/webhook/batchrelease_webhook.go`

验证规则：
1. **创建时验证**：
   - workloadRef 必须存在
   - steps 不能为空
   - replicas 格式正确（数字或百分比）
   - 最后一个 step 必须是 100%

2. **更新时验证**：
   - 不允许修改 workloadRef（immutable）
   - 发布中不允许修改 steps
   - 只允许修改批次状态推进发布

3. **删除时验证**：
   - 记录日志

```go
func (v *BatchReleaseValidator) ValidateCreate(ctx context.Context, obj runtime.Object) error {
    br := obj.(*v1alpha1.BatchRelease)
    
    // 1. 验证 workloadRef
    if br.Spec.WorkloadRef == nil {
        return errors.New("workloadRef is required")
    }
    
    // 2. 验证 steps
    if len(br.Spec.ReleasePlan.Batches) == 0 {
        return errors.New("at least one batch is required")
    }
    
    // 3. 验证最后一步是 100%
    lastBatch := br.Spec.ReleasePlan.Batches[len(br.Spec.ReleasePlan.Batches)-1]
    if lastBatch.CanaryReplicas != "100%" {
        return errors.New("last batch must be 100%")
    }
    
    return nil
}
```

## 四、关键技术点

### 4.1 Finalizer 机制

```go
const FinalizerName = "batchrelease.rollouts.yuyy.com/finalizer"

// 添加 Finalizer
if !controllerutil.ContainsFinalizer(br, FinalizerName) {
    controllerutil.AddFinalizer(br, FinalizerName)
    return r.Update(ctx, br)
}

// 处理删除
if !br.DeletionTimestamp.IsZero() {
    if controllerutil.ContainsFinalizer(br, FinalizerName) {
        // 清理逻辑
        if err := r.cleanupBatchRelease(ctx, br); err != nil {
            return ctrl.Result{}, err
        }
        // 移除 Finalizer
        controllerutil.RemoveFinalizer(br, FinalizerName)
        return r.Update(ctx, br)
    }
    return ctrl.Result{}, nil
}
```
